# Gadai coordination board

Shared board for module builders. **Append-only.** Never edit or delete someone else's line. Add new lines at the bottom of the Log section.

Line format: `[YYYY-MM-DD HH:MM WIB] [module] message`
Modules: `architect`, `contracts`, `agent-core`, `agent-wallet-keeper`, `agent-social-flash`, `agent-flynet-cca`, `web`, `skill-docs`.
Tags to put right after the module:
- `Q:` a question for another module. Name the module, e.g. `Q(contracts):`.
- `A:` an answer. Quote the question's timestamp.
- `DECISION:` a binding decision.
- `CHANGE:` an interface change. Include the old and the new form.
- `BLOCKER:` you can't continue. Say what unblocks you.
- `DONE:` a milestone is finished. Say how to run or verify it.

Rules:
1. Write only in your own directories (SPEC §12). The one exception: the `contracts` module owns the `FEE_DESK_ABI`, `FEE_VAULT_ABI` and `FEE_NOTE_ABI` constants in `shared/src/index.ts`. When you change them, post a `CHANGE:` line here.
2. The interface below is **v1**. Build against it. Any change needs a `CHANGE:` line here first, and after that the SPEC is out of date on that point, so this board wins.
3. Anything a module exports to other modules (SPEC §5) follows the same rule: post `CHANGE:` before you change the signature.
4. No mocks and no fake success paths. If a dependency is missing, throw an error that names the env var or the module.

## Contracts interface v1 (seeded by architect)

Solidity ^0.8.26, Base 8453. Addresses are in `shared/src/index.ts` (`ADDR`). Units: USDC and FeeNote both have 6 decimals, and 1 raw note = 1 raw USDC of face value. A CCA price in Q96 = USDC per note × 2^96.

### FeeDesk.sol (singleton registry and factory)

```solidity
struct CreateLoanParams {
    address borrower;      // current fee beneficiary; receives disbursement + lien back
    bytes32 poolId;
    address feesManager;   // API feesContract ?? initializer
    address creatorToken;  // non-WETH leg of the pool
    uint256 principal;     // USDC raw; = CCA requiredCurrencyRaised
    uint256 faceValue;     // FeeNote supply (raw) = note debt
    uint256 drawLimit;     // USDC raw; Flynet dining line cap
    string noteName;       // "FeeNote GITLAWB #3"
    string noteSymbol;     // "fnGITLAWB3"
}
contract FeeDesk {
    address public owner;    // deployer
    address public keeper;   // Dynamic agent wallet
    address public treasury; // receives dining-draw repayments
    uint256 public loanCount;
    mapping(uint256 => address) public vaultOf;              // loanId => vault (loanId starts at 1)
    mapping(bytes32 => address) public activeVaultByPool;   // one open loan per pool
    mapping(address => bool) public isVault;

    constructor(address keeper_, address treasury_);
    function createLoan(CreateLoanParams calldata p) external /*onlyKeeper*/ returns (uint256 loanId, address vault, address note);
        // require activeVaultByPool[p.poolId]==0, principal>0, faceValue>=principal
        // vault = new FeeVault(...); vault constructor deploys FeeNote (minter = vault)
    function onVaultClosed(bytes32 poolId) external;         // only isVault[msg.sender]; clears activeVaultByPool
    function setKeeper(address) external /*onlyOwner*/;
    function setTreasury(address) external /*onlyOwner*/;

    event LoanCreated(uint256 indexed loanId, address indexed vault, address indexed borrower, address note,
        bytes32 poolId, address feesManager, address creatorToken, uint256 principal, uint256 faceValue, uint256 drawLimit);
    event KeeperSet(address keeper);
    event TreasurySet(address treasury);
}
```

### FeeVault.sol (one per loan: holds the lien, runs the auction, services the loan)

```solidity
enum Status { Created, Pledged, Auction, Active, Released, Cancelled } // uint8 0..5

struct FlashOrder { address swapper; address vault; address recipient; address fromToken; address toToken;
                    uint256 fromAmount; uint256 salt; uint256 deadline; }
// EIP-712 typehash: keccak256("FlashOrder(address swapper,address vault,address recipient,address fromToken,address toToken,uint256 fromAmount,uint256 salt,uint256 deadline)")
// domain: name "DefinitiveFlashAllowance", version "1", chainId block.chainid, verifyingContract 0x5d00000873b6BF41539e6f5365B0Ff7d3c368f78
// (taken from a live Flash /quote orderTypedData on 2026-09-18)

// immutables / views
function desk() view returns (address);  function loanId() view returns (uint256);
function borrower() view returns (address); function poolId() view returns (bytes32);
function feesManager() view returns (address); function creatorToken() view returns (address);
function note() view returns (address); function principal() view returns (uint256);
function faceValue() view returns (uint256); function drawLimit() view returns (uint256);
function status() view returns (uint8); function pledgedAt() view returns (uint64);
function auction() view returns (address); function drawn() view returns (uint256);
function drawDebt() view returns (uint256);
function debtOutstanding() view returns (uint256); // noteSupply + drawDebt - min(usdcBal, noteSupply + drawDebt)
function canRelease() view returns (bool);         // status==Active && usdcBal >= noteSupply + drawDebt
function approvedHash(bytes32) view returns (bool);

// lifecycle
function confirmPledge();            // anyone. Created→Pledged. require getShares(poolId,this)>0 && getShares(poolId,borrower)==0
function startAuction(uint64 startBlock, uint64 endBlock, uint64 claimBlock, uint256 tickSpacingQ96,
                      uint256 floorPriceQ96, bytes stepsData) returns (address auction);
                                     // onlyKeeper. Pledged→Auction. params: currency=USDC, tokensRecipient=fundsRecipient=this,
                                     // validationHook=0, requiredCurrencyRaised=principal; factory.create(note, faceValue, abi.encode(params), bytes32(loanId));
                                     // note.mint(auction, faceValue); auction.onTokensReceived()
function disburse() returns (uint256 amount); // onlyKeeper. Auction→Active. block>=endBlock && isGraduated; sweepCurrency();
                                     // sweepUnsoldTokens()+burn; send exactly the swept USDC to borrower
function cancel();                   // Created: keeper. Pledged: keeper, or anyone after pledgedAt+1 days.
                                     // Auction: anyone if block>=endBlock && !isGraduated (sweep+burn unsold notes).
                                     // Then: updateBeneficiary(poolId, borrower) if shares>0; send all WETH/creatorToken/USDC to borrower;
                                     // desk.onVaultClosed. →Cancelled

// servicing (Active unless noted)
function collect() returns (uint128 fees0, uint128 fees1); // anyone; Pledged|Auction|Active; feesManager.collectFees(poolId)
function swapWethToUsdc(bytes swapData, uint256 amountIn, uint256 minUsdcOut) returns (uint256 usdcOut);
                                     // onlyKeeper. WETH.approve(SwapProxy 0x0000000085E1…affad, amountIn); call(swapData); approve 0;
                                     // require USDC delta >= minUsdcOut
function approveFlash(address token, uint256 amount);      // onlyKeeper; token ∈ {creatorToken, WETH}; approve FLASH_SETTLEMENT
function authorizeFlashOrder(FlashOrder o) returns (bytes32 digest);
                                     // onlyKeeper; require o.swapper==this && o.recipient==this && o.toToken==USDC
                                     // && o.fromToken ∈ {creatorToken,WETH} && o.deadline>block.timestamp; approvedHash[digest]=true
function authorizeFlashCancel(string orderId) returns (bytes32 digest);
                                     // onlyKeeper; digest = EIP-191 personal_sign hash of
                                     // "Definitive Flash v1 — Cancel Order\nOrder: <orderId>" (UTF-8, em dash U+2014)
function isValidSignature(bytes32 hash, bytes signature) view returns (bytes4); // 0x1626ba7e iff approvedHash[hash]
function sendTokenLegToKeeper(address token, uint256 amount); // onlyKeeper; fallback when Flash rejects 1271 funders
function addDraw(uint256 usdcAmount); // onlyKeeper; Active; drawn+amt <= drawLimit; drawn+=, drawDebt+=
function payDesk() returns (uint256 paid); // anyone; pays treasury min(drawDebt, usdcBal - noteSupply) (notes are senior)
function redeem(uint256 notes);      // anyone holding notes; Active|Released; burn notes, pay USDC 1:1 (needs balance)
function release();                  // anyone; require canRelease(); payDesk(); updateBeneficiary(poolId, borrower);
                                     // reset Flash approvals; send WETH, creatorToken, USDC surplus (bal - noteSupply) to borrower;
                                     // desk.onVaultClosed. →Released. USDC for unredeemed notes stays for redeem()

event Pledged(address indexed borrower, uint256 shares);
event AuctionStarted(address indexed auction, uint64 startBlock, uint64 endBlock, uint256 floorPriceQ96, uint256 faceValue);
event Disbursed(address indexed borrower, uint256 amount);
event Collected(uint256 fees0, uint256 fees1);
event Swapped(uint256 wethIn, uint256 usdcOut);
event FlashOrderAuthorized(bytes32 indexed digest, address fromToken, uint256 fromAmount);
event TokenLegSent(address indexed keeper, address token, uint256 amount);
event Drawn(uint256 amount, uint256 drawDebt);
event DeskPaid(uint256 amount);
event Redeemed(address indexed holder, uint256 notes);
event Released(address indexed borrower);
event Cancelled(uint8 fromStatus);
```
